Settings card Pattern

The .dirty-section pattern wraps a card with a title, fields, and a save row. Add .is-dirty to the section when any input changes to signal unsaved state.

Default

Notification preferences

Email alerts

Send me an email for important updates.

<div class="dirty-section" id="my-section">
  <h3 class="dirty-section__title">Notification preferences</h3>
  <div class="dirty-section__card">
    <div class="dirty-section__card-body">
      <div class="dirty-section__field-row">
        <div class="dirty-section__field-info">
          <p class="dirty-section__field-label">Email alerts</p>
          <p class="dirty-section__field-description">
            Send me an email for important updates.
          </p>
        </div>
        <div class="dirty-section__control">
          <label class="ds-toggle">
            <input type="checkbox" checked>
            <span class="ds-toggle__track"></span>
          </label>
        </div>
      </div>
      <div class="dirty-section__field">
        <label class="dirty-section__field-label">Display name</label>
        <input class="dirty-section__input" type="text" value="Jane Smith">
      </div>
    </div>
    <div class="dirty-section__save-row">
      <button class="btn-new btn-new--secondary btn-new--small">Cancel</button>
      <button class="btn-new btn-new--primary btn-new--small">Save changes</button>
    </div>
  </div>
</div>

<script>
// Mark dirty on any change
const section = document.getElementById('my-section');
section.querySelectorAll('input, textarea').forEach(el =>
  el.addEventListener('change', () => section.classList.add('is-dirty'))
);
</script>

<div class="ds-pane" id="tab-settings">Settings content</div>
<div class="ds-pane" id="tab-history">History content</div>

<script>
document.getElementById('demo-tabs').addEventListener('click', e => {
  const tab = e.target.closest('.ds-tab');
  if (!tab) return;
  document.querySelectorAll('.ds-tab').forEach(t => t.classList.remove('active'));
  document.querySelectorAll('.ds-pane').forEach(p => p.classList.remove('active'));
  tab.classList.add('active');
  document.getElementById('tab-' + tab.dataset.tab).classList.add('active');
});
</script>
